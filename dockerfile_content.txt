L1: # ─────────────────────────────────────────────────────────────────────────────
L2: # Dashboard ECCSA — el kiosco de la oficina.
L3: #
L4: # Dos etapas, como el Dockerfile de Field:
L5: #   1. node:20-alpine compila el build estático de SvelteKit,
L6: #   2. python:3.11-slim + nginx + supervisor corre el resultado.
L7: #
L8: # Las dos bases salen de mirror.gcr.io (espejo pull-through de Docker Hub):
L9: # Docker Hub y ECR Public aplican cuotas por IP y ya tumbaron la CI dos veces.
L10: # El espejo sirve el MISMO manifiesto, sin esa cuota.
L11: #
L12: # El Dockerfile NO lleva credenciales: entran por --env-file al hacer `docker
L13: # run` (ver deploy/run_container.ps1) y el entrypoint las convierte en
L14: # /app/api/secretos_local.py, que está en .gitignore.
L15: # ─────────────────────────────────────────────────────────────────────────────
L16: FROM mirror.gcr.io/library/node:20-alpine AS build
L17: WORKDIR /app
L18: COPY package*.json ./
L19: RUN npm ci
L20: COPY . .
L21: ARG BUILD_ID=dev
L22: RUN echo "$BUILD_ID" > static/build.txt && npm run build
L23: 
L24: FROM mirror.gcr.io/library/python:3.11-slim
L25: 
L26: # nginx para servir la pantalla y los medios; supervisor para los 3 procesos.
L27: # `curl` lo usa el HEALTHCHECK del final.
L28: RUN apt-get update && apt-get install -y --no-install-recommends \
L29:         nginx supervisor curl \
L30:     && rm -rf /var/lib/apt/lists/*
L31: 
L32: # pymssql necesita FreeTDS (el driver de SQL Server).
L33: COPY api/requirements.txt /tmp/requirements.txt
L34: RUN apt-get update && apt-get install -y --no-install-recommends \
L35:         freetds-dev gcc \
L36:     && pip install --no-cache-dir -r /tmp/requirements.txt \
L37:     && apt-get purge -y gcc freetds-dev \
L38:     && apt-get autoremove -y \
L39:     && rm -rf /var/lib/apt/lists/*
L40: 
L41: # App: build estático, API, worker, configuración de nginx y supervisor.
L42: COPY --from=build /app/build /app/build
L43: COPY api/    /app/api/
L44: COPY tz_mx.py /app/tz_mx.py
L45: COPY worker/ /app/worker/
L46: # Las versiones que anuncia /api/shell/state (contrato del ECCSA-Shell):
L47: # app.version sale de package.json y shell.version del ECCSA_SHELL_VERSION que
L48: # estampa `sync_shell.py`. Sin estos dos archivos el endpoint respondía "?" y la
L49: # pantalla no podía decir qué build tenía arriba.
L50: COPY package.json          /app/package.json
L51: COPY ECCSA_SHELL_VERSION   /app/ECCSA_SHELL_VERSION
L52: COPY docker/nginx.conf /etc/nginx/sites-available/default
L53: COPY docker/nginx.conf /etc/nginx/conf.d/default.conf
L54: COPY docker/supervisor/ /app/docker/supervisor/
L55: COPY docker/entrypoint.sh /app/docker/entrypoint.sh
L56: RUN chmod +x /app/docker/entrypoint.sh
L57: 
L58: # El build se identifica con un hash: la pantalla lo consulta en
L59: # /api/dashboard/version y se recarga sola si cambió tras un deploy.
L60: RUN date -u +%Y%m%d-%H%M%S > /app/BUILD
L61: 
L62: RUN mkdir -p /data/media/thumbs /data/media/wallpapers /data/media/avatars \
L63:              /var/log/supervisor /var/cache/nginx /run/nginx
L64: VOLUME ["/data"]
L65: 
L66: EXPOSE 80
L67: 
L68: # El healthcheck pregunta al backend, no al puerto: nginx puede estar arriba
L69: # mientras la API no. 503 = degradado, y el contenedor se reinicia solo.
L70: HEALTHCHECK --interval=30s --timeout=5s --start-period=40s --retries=3 \
L71:   CMD curl -fsS http://127.0.0.1:8100/healthz || exit 1
L72: 
L73: CMD ["/app/docker/entrypoint.sh"]
